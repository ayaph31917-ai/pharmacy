        }

        function filterCategory(category, btn) {
            document.querySelectorAll('.cat-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            if(category === 'الكل' || category === 'all') {
                renderProducts(storeData.products);
            } else {
                const filtered = storeData.products.filter(p => p.category === category);
                renderProducts(filtered);
            }
        }

        function switchTab(tabId, element) {
            document.querySelectorAll('.page-tab').forEach(tab => tab.classList.remove('active'));
            document.querySelectorAll('.nav-link').forEach(link => link.classList.remove('active'));
            document.getElementById(tabId).classList.add('active');
            element.classList.add('active');
        }

        function addToCart(title, price) {
            cart.push({ title, price });
            document.getElementById('cart-count').innerText = cart.length;
            alert(`تمت إضافة "${title}" إلى سلة التسوق`);
        }

        function openCart() {
            if(cart.length === 0) {
                alert("سلة التسوق فارغة حالياً!");
                return;
            }
            let text = `مرحباً دكتورة آية، أود طلب المواد التالية من الصيدلية:\n\n`;
            cart.forEach((item, index) => {
                text += `${index+1}. ${item.title} (${item.price})\n`;
            });
            const url = `https://wa.me/${storeData.whatsapp}?text=${encodeURIComponent(text)}`;
            window.open(url, '_blank');
        }

        function sendProductRequest(e) {
            e.preventDefault();
            const name = document.getElementById('req-name').value;
            const prod = document.getElementById('req-product').value;
            const notes = document.getElementById('req-notes').value;
            const msg = `طلب توفير منتج جديد:\nالاسم: ${name}\nالمنتج المطلوب: ${prod}\nملاحظات: ${notes}`;
            window.open(`https://wa.me/${storeData.whatsapp}?text=${encodeURIComponent(msg)}`, '_blank');
        }

        function sendPrescription(e) {

            window.open(`https://wa.me/${storeData.whatsapp}?text=${encodeURIComponent(msg)}`, '_blank');
        }
