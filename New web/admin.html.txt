        function getStoredData() {
            const data = localStorage.getItem('phy2li_store_data');
            return data ? JSON.parse(data) : defaultData;
        }

        let storeData = getStoredData();

        function login() {
            const u = document.getElementById('adminUser').value;
            const p = document.getElementById('adminPass').value;

            if(u === (storeData.username || "admin") && p === (storeData.password || "123")) {
                document.getElementById('loginPanel').style.display = 'none';
                document.getElementById('dashboardPanel').style.display = 'block';
                loadAdminData();
            } else {
                alert("اسم المستخدم أو كلمة المرور غير صحيحة!");
            }
        }

        function logout() {
            document.getElementById('loginPanel').style.display = 'block';
            document.getElementById('dashboardPanel').style.display = 'none';
        }

        function showPanel(panelId, btn) {
            document.querySelectorAll('.section-panel').forEach(p => p.classList.remove('active'));
            document.querySelectorAll('.admin-btn').forEach(b => b.classList.remove('active'));
            document.getElementById(panelId).classList.add('active');
            btn.classList.add('active');
        }

        function renderProductsTable() {
            const tbody = document.getElementById('productsTable');
            tbody.innerHTML = '';
            storeData.products.forEach(p => {
                tbody.innerHTML += `
                    <tr>
                        <td>${p.title}</td>
                        <td>${p.category}</td>
                        <td>${p.price}</td>
                        <td><button class="action-btn-del" onclick="deleteProduct(${p.id})">حذف</button></td>
                    </tr>
                `;
            });
        }

        function addProduct() {
            const title = document.getElementById('p-title').value;
            const category = document.getElementById('p-cat').value;
            const price = document.getElementById('p-price').value;
            const img = document.getElementById('p-img').value;
            const desc = document.getElementById('p-desc').value;

            if(!title || !price) {
                alert("يرجى أدخال اسم المنتج والسعر على الأقل");
                return;
            }

            const newProd = {
                id: Date.now(),
                title,
                category: category || "عام",
                price,
                img,
                desc
            };

            storeData.products.push(newProd);
            localStorage.setItem('phy2li_store_data', JSON.stringify(storeData));
            renderProductsTable();
            alert("تمت إضافة المنتج بنجاح!");
        }

        function deleteProduct(id) {
            storeData.products = storeData.products.filter(p => p.id !== id);
            localStorage.setItem('phy2li_store_data', JSON.stringify(storeData));
            renderProductsTable();
        }

        function renderSkinTable() {
            const tbody = document.getElementById('skinTable');
            tbody.innerHTML = '';
            storeData.skinProblems.forEach(s => {
                tbody.innerHTML += `
                    <tr>
                        <td>${s.title}</td>
                        <td>${s.solution}</td>
                        <td><button class="action-btn-del" onclick="deleteSkin(${s.id})">حذف</button></td>
                    </tr>
                `;
            });
        }

        function addSkinProblem() {
            const title = document.getElementById('s-title').value;
            const solution = document.getElementById('s-solution').value;

            if(!title || !solution) {
                alert("يرجى ملء جميع الحقول");
                return;
            }

            storeData.skinProblems.push({ id: Date.now(), title, solution });
            localStorage.setItem('phy2li_store_data', JSON.stringify(storeData));
            renderSkinTable();
            alert("تمت الإضافة بنجاح!");
        }

        function deleteSkin(id) {
            storeData.skinProblems = storeData.skinProblems.filter(s => s.id !== id);
            localStorage.setItem('phy2li_store_data', JSON.stringify(storeData));
            renderSkinTable();
        }
